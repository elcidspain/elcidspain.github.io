async function runTwin(hud) { const C = window.Cesium; window.CESIUM_BASE_URL = "https://cesium.com/downloads/cesiumjs/releases/1.125/Build/Cesium/"; const ion = new URLSearchParams(location.search).get("ion") || localStorage.getItem("CESIUM_ION_TOKEN"); if (ion) { C.Ion.defaultAccessToken = ion; localStorage.setItem("CESIUM_ION_TOKEN", ion); } const [geo, flight] = await Promise.all([ fetch("./AUMARA_WORLD_GEOREFERENCE_v1.json").then((r) => r.json()), fetch("./world/flight-path.json").then((r) => r.json()), ]); const lon = geo.localOrigin.wgs84.longitude; const lat = geo.localOrigin.wgs84.latitude; const viewer = new C.Viewer("c", { animation: false, timeline: false, baseLayerPicker: false, geocoder: C.IonGeocodeProviderType.GOOGLE, homeButton: false, sceneModePicker: false, navigationHelpButton: false, fullscreenButton: false, infoBox: false, selectionIndicator: false, terrainProvider: new C.EllipsoidTerrainProvider(), baseLayer: false, shadows: true, }); viewer.scene.globe.show = false; viewer.scene.skyAtmosphere.show = true; viewer.scene.fog.enabled = true; viewer.scene.light = new C.SunLight({ intensity: 2.0 }); viewer.scene.globe.enableLighting = true; let tilesOn = false; try { const googleTiles = await C.createGooglePhotorealistic3DTileset({ onlyUsingWithGoogleGeocoder: true, }); try { googleTiles.customShader = new C.CustomShader({ lightingModel: C.LightingModel.PBR, uniforms: { u_exposure: { type: C.UniformType.FLOAT, value: 1.12 }, }, fragmentShaderText: ` void fragmentMain(FragmentInput fsInput, inout czm_modelMaterial material) { material.diffuse *= u_exposure; material.roughness = clamp(material.roughness, 0.35, 0.92); } `, }); } catch (shaderErr) {} viewer.scene.primitives.add(googleTiles); tilesOn = true; hud.textContent = "Earth tiles on"; } catch (e) { try { const tiles = await C.Cesium3DTileset.fromIonAssetId(2275207); viewer.scene.primitives.add(tiles); tilesOn = true; hud.textContent = "Ion 2275207 on"; } catch (e2) { const esri = await C.ArcGisMapServerImageryProvider.fromUrl(ESRI); viewer.imageryLayers.addImageryProvider(esri); viewer.scene.globe.show = true; viewer.terrainProvider = await C.createWorldTerrainAsync(); hud.textContent = "ESRI fallback · ion token later"; } } const origin = C.Cartesian3.fromDegrees(lon, lat, 0); const modelMatrix = C.Transforms.eastNorthUpToFixedFrame(origin); const glb = await C.Model.fromGltfAsync({ url: "./world/aumara-site.glb", modelMatrix, scale: 1.0, }); glb.customShader = new C.CustomShader({ lightingModel: C.LightingModel.PBR, uniforms: { u_sunBoost: { type: C.UniformType.FLOAT, value: 1.25, }, }, fragmentShaderText: ` void fragmentMain(FragmentInput fsInput, inout czm_modelMaterial material) { material.diffuse = mix(material.diffuse, vec3(0.42, 0.36, 0.28), 0.08); material.roughness = clamp(material.roughness * 0.9, 0.4, 0.88); material.metallic = min(material.metallic, 0.08); material.emissive *= 0.0; material.diffuse *= u_sunBoost; } `, }); viewer.scene.primitives.add(glb); const wps = flight.waypoints; const start = C.JulianDate.now(); const intro = 10; const dur = flight.duration_s_default || 48; const pos = new C.SampledPositionProperty(); pos.setInterpolationOptions({ interpolationDegree: 2, interpolationAlgorithm: C.HermitePolynomialApproximation, }); pos.addSample(start, C.Cartesian3.fromDegrees(lon, lat, 2.4e6)); wps.forEach((wp, i) => { const t = C.JulianDate.addSeconds( start, intro + (i / (wps.length - 1)) * dur, new C.JulianDate(), ); const agl = Math.min(4, Math.max(3.5, wp.target_agl_m || 3.75)); pos.addSample( t, C.Cartesian3.fromDegrees( wp.wgs84.longitude, wp.wgs84.latitude, (wp.local.ground_z || 0) + agl, ), ); }); viewer.clock.startTime = start.clone(); viewer.clock.stopTime = C.JulianDate.addSeconds( start, intro + dur, new C.JulianDate(), ); viewer.clock.currentTime = start.clone(); viewer.clock.clockRange = C.ClockRange.LOOP_STOP; viewer.clock.shouldAnimate = true; viewer.clock.onTick.addEventListener(() => { const p = pos.getValue(viewer.clock.currentTime); if (!p) return; const t2 = C.JulianDate.addSeconds(viewer.clock.currentTime, 0.4, new C.JulianDate()); const n = pos.getValue(t2) || p; const dir = C.Cartesian3.normalize( C.Cartesian3.subtract(n, p, new C.Cartesian3()), new C.Cartesian3(), ); const up = C.Ellipsoid.WGS84.geodeticSurfaceNormal(p, new C.Cartesian3()); const right = C.Cartesian3.normalize( C.Cartesian3.cross(dir, up, new C.Cartesian3()), new C.Cartesian3(), ); const look = C.Matrix3.multiplyByVector( C.Matrix3.fromQuaternion(C.Quaternion.fromAxisAngle(right, -0.38)), dir, new C.Cartesian3(), ); viewer.camera.setView({ destination: p, orientation: { direction: look, up } }); const carto = C.Cartographic.fromCartesian(p); let ground = 0; try { const sampled = viewer.scene.sampleHeight(carto); if (typeof sampled === "number") ground = sampled; } catch {} const elapsed = C.JulianDate.secondsDifference(viewer.clock.currentTime, start); const wp = elapsed < intro ? 0 : Math.min(27, Math.round(((elapsed - intro) / dur) * 27)); hud.textContent = `AUMARA WP ${wp}/27 AGL ${(carto.height - ground).toFixed(2)} m ${tilesOn ? "ION" : "ESRI"} GLB`; }); }
